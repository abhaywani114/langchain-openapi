    class AsyncHTTPExecutor:
    """Generic asynchronous HTTP executor for executing OpenAPI operations.

    The executor is completely decoupled from AI agents and LangChain interfaces.

    Attributes:
        base_url: Optional default base URL for API endpoints.
        client: Optional injected httpx.AsyncClient instance.
        provider: RequestProvider instance for authentication and mutation.
        pipeline: MiddlewarePipeline for processing request/response lifecycle.
        timeout: Execution timeout in seconds.
    """

    def __init__(
        self,
        base_url: str | None = None,
        client: httpx.AsyncClient | None = None,
        provider: RequestProvider | None = None,
        auth: AuthProvider | None = None,
        middleware: Sequence[Middleware] | None = None,
        timeout: float = 30.0,
    ) -> None:
        """Initialize AsyncHTTPExecutor.

        Args:
            base_url: Optional base URL prefix for operations.
            client: Optional httpx.AsyncClient for dependency injection.
            provider: RequestProvider instance for request mutation & authentication.
            auth: Legacy AuthProvider instance for backwards compatibility.
            middleware: Sequence of Middleware objects to process HTTP requests.
            timeout: Request timeout duration in seconds (default 30.0s).
        """
        self.base_url = base_url
        self._client = client

        if provider is not None:
            self.provider = provider
        elif auth is not None:
            self.provider = AuthProviderAdapter(auth)
        else:
            self.provider = NoAuthProvider()

        self.pipeline = MiddlewarePipeline(middleware)
        self.timeout = timeout
        self._builder = RequestBuilder(base_url=base_url)
        self._parser = ResponseParser()

    async def execute(
        self,
        operation: Operation,
        arguments: dict[str, Any],
        base_url_override: str | None = None,
    ) -> ResponseData:
        """Execute an OpenAPI Operation with the supplied arguments.

        Args:
            operation: Target Operation model.
            arguments: Input arguments dictionary.
            base_url_override: Optional base URL override.

        Returns:
            Normalized ResponseData object containing status, headers, and parsed body.

        Raises:
            RequestValidationError: If path parameters are missing.
            AuthenticationError: If authentication configuration fails.
            ProviderError: If a request provider fails.
            ExecutionTimeoutError: If request execution times out.
            HTTPExecutionError: If network or transport error occurs.
            ResponseParsingError: If response parsing fails.
        """
        # 1. Build HTTP request metadata
        built_req = self._builder.build(
            operation=operation,
            arguments=arguments,
            base_url_override=base_url_override,
        )

        start_time = time.monotonic()

        # 2. Perform network request using client or dedicated context
        try:
            if self._client is not None:
                response = await self._send_request(self._client, built_req)
            else:
                async with httpx.AsyncClient(timeout=self.timeout) as client:
                    response = await self._send_request(client, built_req)
        except httpx.TimeoutException as exc:
            logger.error(
                "Request timed out for %s after %.1fs",
                sanitize_request_log(built_req.method, built_req.url),
                self.timeout,
            )
            raise ExecutionTimeoutError(
                f"HTTP request to '{built_req.url}' timed out after {self.timeout}s."
            ) from exc
        except (
            RequestValidationError,
            AuthenticationError,
            ProviderError,
            ResponseParsingError,
        ):
            raise
        except httpx.HTTPError as exc:
            logger.error(
                "HTTP execution failed for %s: %s",
                sanitize_request_log(built_req.method, built_req.url),
                exc,
            )
            raise HTTPExecutionError(
                f"HTTP request execution failed for '{built_req.url}': {exc}"
            ) from exc

        elapsed = time.monotonic() - start_time
        logger.info(
            "Executed HTTP %s -> Status %d (%.3fs)",
            sanitize_request_log(built_req.method, built_req.url),
            response.status_code,
            elapsed,
        )

        # 3. Parse and return normalized response
        return self._parser.parse(response)

    async def _send_request(
        self, client: httpx.AsyncClient, req_data: Any
    ) -> httpx.Response:
        kwargs: dict[str, Any] = {
            "method": req_data.method,
            "url": req_data.url,
            "headers": req_data.headers,
            "params": req_data.params,
            "json": req_data.json_body,
            "timeout": self.timeout,
        }

        request = client.build_request(**kwargs)

        # Apply cookies if present on BuiltRequest
        if req_data.cookies:
            cookie_hdr = "; ".join(f"{k}={v}" for k, v in req_data.cookies.items())
            request.headers["Cookie"] = cookie_hdr

        # Apply request provider mutations (auth, headers, cookies, etc.)
        request = await self.provider.apply(request)

        # Execute request through middleware pipeline down to transport
        async def transport_call(req: httpx.Request) -> httpx.Response:
            return await client.send(req)

        return await self.pipeline.execute(request, transport_call)

    class RequestBuilder:
    """Builder for translating Operation models into BuiltRequest instances."""

    def __init__(self, base_url: str | None = None) -> None:
        """Initialize RequestBuilder.

        Args:
            base_url: Default base URL for API requests.
        """
        self.base_url = base_url.rstrip("/") if base_url else ""

    def build(
        self,
        operation: Operation,
        arguments: dict[str, Any],
        base_url_override: str | None = None,
    ) -> BuiltRequest:
        """Build an HTTP request representation for an operation and arguments.

        Args:
            operation: Internal Operation model.
            arguments: Input arguments dictionary.
            base_url_override: Optional base URL override.

        Returns:
            A BuiltRequest object containing request parameters and payload.

        Raises:
            RequestValidationError: If a required path parameter is missing.
        """
        method = operation.method.value.upper()
        base_url = base_url_override.rstrip("/") if base_url_override else self.base_url

        path_template = operation.path
        query_params: dict[str, Any] = {}
        headers: dict[str, str] = {}
        cookies: dict[str, str] = {}
        consumed_args: set[str] = set()

        # Handle special paginate/__paginate__ argument for pagination middleware
        if arguments.get("paginate") or arguments.get("__paginate__"):
            headers["X-LangChain-Paginate"] = "true"
            consumed_args.add("paginate")
            consumed_args.add("__paginate__")

        # 1. Process operation parameters
        for param in operation.parameters:
            name = param.name
            val = arguments.get(name)

            if param.location == ParameterLocation.PATH:
                if val is None:
                    raise RequestValidationError(
                        f"Missing required path parameter '{name}' "
                        f"for operation '{operation.name}'."
                    )
                placeholder = f"{{{name}}}"
                path_template = path_template.replace(placeholder, str(val))
                consumed_args.add(name)

            elif param.location == ParameterLocation.QUERY:
                if val is not None:
                    query_params[name] = val
                    consumed_args.add(name)

            elif param.location == ParameterLocation.HEADER:
                if val is not None:
                    headers[name] = str(val)
                    consumed_args.add(name)

            elif param.location == ParameterLocation.COOKIE:
                if val is not None:
                    cookies[name] = str(val)
                    consumed_args.add(name)

        # 2. Construct final URL
        if base_url:
            full_url = f"{base_url}/{path_template.lstrip('/')}"
        else:
            full_url = path_template

        # 3. Process Request Body
        json_body: Any = None
        if operation.request_body:
            if "body" in arguments:
                json_body = arguments["body"]
            else:
                unconsumed = {
                    k: v for k, v in arguments.items() if k not in consumed_args
                }
                if unconsumed:
                    json_body = unconsumed

            if json_body is not None and "Content-Type" not in headers:
                headers["Content-Type"] = "application/json"

        return BuiltRequest(
            method=method,
            url=full_url,
            headers=headers,
            params=query_params,
            json_body=json_body,
            cookies=cookies,
        )

    class ResponseParser:
    """Parser for attempting JSON decoding and standardizing HTTP responses."""

    def parse(self, response: httpx.Response) -> ResponseData:
        """Parse an httpx.Response into a normalized ResponseData object.

        Args:
            response: The raw httpx.Response instance.

        Returns:
            A ResponseData object containing status, headers, and parsed body.

        Raises:
            ResponseParsingError: If reading response content fails.
        """
        try:
            status_code = response.status_code
            headers = dict(response.headers)

            # Attempt JSON decoding if response content exists
            if response.content:
                try:
                    body: Any = response.json()
                    logger.debug("Successfully decoded response body as JSON")
                except Exception:
                    body = response.text
                    logger.debug("Response is not JSON, decoded as plain text")
            else:
                body = None

            return ResponseData(
                status_code=status_code,
                headers=headers,
                body=body,
                raw=response,
            )
        except Exception as exc:
            logger.error("Failed to parse response: %s", exc)
            raise ResponseParsingError(
                f"Catastrophic error parsing HTTP response: {exc}"
            ) from exc
